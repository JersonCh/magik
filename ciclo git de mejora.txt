# 1. Asegurarte de estar en develop actualizado
git checkout develop
git pull origin develop

# 2. Crear tu rama de trabajo para la tarea específica
git checkout -b feature/cloudflare-d1

# 3. Desarrollamos los cambios y probamos (build, dev, etc.)
# ...

# 4. Hacemos commit y subimos la rama
git add .
git commit -m "feat: implementar esquema D1 y capa db.ts"
git push -u origin feature/cloudflare-d1

# 5. Integramos a develop (vía merge local o Pull Request en GitHub)
git checkout develop
git merge feature/cloudflare-d1
git push origin develop

# 6. Cuando develop tiene un conjunto de mejoras 100% probadas y listas:
git checkout main
git merge develop
git push origin main  # -> ¡Esto actualiza el sitio en producción!
